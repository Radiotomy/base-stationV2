# Cadence engine dependencies.
# torch / torchaudio are installed separately in the Dockerfile from the CUDA
# 12.1 index and are deliberately absent here.

# Meta's AudioCraft supplies MusicGen + the melody chroma conditioner we patch.
# Pinned: audiocraft's conditioner internals are what our chord injection wraps,
# and an unpinned bump could rename the method we patch.
audiocraft==1.3.0

# MUST be pinned. Left unpinned, audiocraft resolves the newest transformers,
# which enforces a torch >= 2.2 minimum and — instead of erroring at install —
# makes is_torch_available() return False at runtime. The symptom is the
# misleading "T5EncoderModel requires the PyTorch library but it was not found",
# raised on the first render while /health still correctly reports cuda: true.
# audiocraft pins torch to 2.1.0, so transformers must stay on a 2.1-era release.
# 4.40.2 also predates the torchao/transformers versioning that produced the
# "module torch has no attribute int1" crash on the Coda Space.
transformers==4.40.2

# Written with soundfile rather than torchaudio.save — torchcodec fails to load
# on the HF Space image.
soundfile==0.12.1
numpy<2.0

fastapi==0.115.0
uvicorn==0.30.6
pydantic==2.9.2
